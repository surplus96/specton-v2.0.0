"""Local customer identities; isolated from Specton accounts and preview tokens."""
import hashlib
import hmac
import json
import os
import re
import secrets
from urllib.parse import urlsplit

import psycopg
from psycopg.rows import dict_row
from psycopg.types.json import Jsonb
from fastapi import HTTPException, Request, Response
from pydantic import BaseModel, ConfigDict, Field

COOKIE = 'specton_customer_session'
DDL = '''
CREATE TABLE _specton_users (
 username text PRIMARY KEY, password_hash text NOT NULL, roles jsonb NOT NULL,
 active boolean NOT NULL DEFAULT true);
CREATE TABLE _specton_sessions (
 token_hash text PRIMARY KEY, username text NOT NULL REFERENCES _specton_users,
 csrf text NOT NULL, expires_at timestamptz NOT NULL);
CREATE INDEX ON _specton_sessions(expires_at);
CREATE TABLE _specton_login_attempts (
 bucket text PRIMARY KEY, attempts integer NOT NULL, resets_at timestamptz NOT NULL);
'''


def password_hash(password, salt=None):
    salt = salt or secrets.token_hex(16)
    key = hashlib.scrypt(password.encode(), salt=bytes.fromhex(salt), n=16384, r=8, p=1)
    return 'scrypt$' + salt + '$' + key.hex()


def verify_password(password, encoded):
    try:
        kind, salt, _ = encoded.split('$')
        return kind == 'scrypt' and hmac.compare_digest(password_hash(password, salt), encoded)
    except (ValueError, TypeError):
        return False


def roles_in(config):
    operations = list(config['operations'])
    for name in ('domain', 'inventory', 'commerce'):
        operations.extend((config.get(name) or {}).get('operations', []))
    return {r for op in operations for r in op['actor_roles']}


def save_user(conn, config, username, password, roles):
    if not re.fullmatch(r'[a-z0-9][a-z0-9_.@-]{2,63}', username):
        raise ValueError('아이디는 소문자·숫자·_.@-로 3~64자여야 합니다.')
    if not 12 <= len(password) <= 128 or not roles or not set(roles) <= roles_in(config):
        raise ValueError('12~128자 비밀번호와 승인 명세의 역할이 필요합니다.')
    conn.execute('INSERT INTO _specton_users VALUES (%s,%s,%s,true) ON CONFLICT(username) '
                 'DO UPDATE SET password_hash=EXCLUDED.password_hash,roles=EXCLUDED.roles,active=true',
                 (username, password_hash(password), Jsonb(sorted(set(roles)))))
    conn.execute('DELETE FROM _specton_sessions WHERE username=%s', (username,))


class Login(BaseModel):
    model_config = ConfigDict(extra='forbid')
    username: str = Field(min_length=1, max_length=64)
    password: str = Field(min_length=1, max_length=128)


def install(app, config, dsn):
    tokens = json.loads(os.environ.get('APP_TOKEN_HASHES', '{}'))
    mode = os.environ.get('APP_AUTH_MODE', 'token' if tokens else 'session')
    if mode not in {'token', 'session'} or (mode == 'token' and not tokens):
        raise ValueError('Explicit token configuration or customer session mode is required')
    origin = os.environ.get('APP_ORIGIN', 'http://localhost:8080')
    parsed = urlsplit(origin)
    secure = os.environ.get('APP_COOKIE_SECURE', 'true').lower() == 'true'
    if mode == 'session' and (
        parsed.scheme not in {'http', 'https'} or not parsed.hostname or parsed.path
        or parsed.query or parsed.fragment or parsed.username or parsed.password
        or (parsed.scheme == 'http' and (secure or parsed.hostname not in {'localhost','127.0.0.1','::1'}))
        or (parsed.scheme == 'https' and not secure)
    ):
        raise ValueError('Use an HTTPS origin with secure cookies; HTTP is local-only and explicit')
    approved_roles = roles_in(config)
    dummy = password_hash('unused-dummy-password', '0' * 32)

    def same_origin(request):
        if request.headers.get('origin') != origin:
            raise HTTPException(403, detail='ORIGIN_REJECTED')

    def identity(request):
        if mode == 'token':
            value = request.headers.get('authorization', '')
            key = hashlib.sha256(value[7:].encode()).hexdigest() if value.startswith('Bearer ') else ''
            if key not in tokens:
                raise HTTPException(401, detail='AUTH_REQUIRED')
            return {'username':key, 'roles':list(set(tokens[key]) & approved_roles), 'csrf':None}
        value = request.cookies.get(COOKIE, '')
        if not re.fullmatch(r'[A-Za-z0-9_-]{43}', value):
            raise HTTPException(401, detail='AUTH_REQUIRED')
        with psycopg.connect(dsn, row_factory=dict_row) as conn:
            row = conn.execute('SELECT u.username,u.roles,s.csrf FROM _specton_sessions s '
                               'JOIN _specton_users u USING(username) WHERE s.token_hash=%s '
                               'AND s.expires_at>now() AND u.active',
                               (hashlib.sha256(value.encode()).hexdigest(),)).fetchone()
        if not row:
            raise HTTPException(401, detail='AUTH_REQUIRED')
        row['roles'] = sorted(set(row['roles']) & approved_roles)
        return row

    def principal(request, allowed):
        user = identity(request)
        if mode == 'session' and request.method not in {'GET','HEAD','OPTIONS'}:
            same_origin(request)
            if not hmac.compare_digest(request.headers.get('x-csrf-token', '').encode(), user['csrf'].encode()):
                raise HTTPException(403, detail='CSRF_REJECTED')
        if not set(user['roles']) & set(allowed):
            raise HTTPException(403, detail='FORBIDDEN')
        return user['username']

    @app.middleware('http')
    async def private_responses(request, call_next):
        response = await call_next(request)
        response.headers['Cache-Control'] = 'no-store'
        return response

    @app.get('/auth/session')
    def session(request: Request, response: Response):
        response.headers['Cache-Control'] = 'no-store'
        try:
            user = identity(request)
        except HTTPException as exc:
            if exc.status_code != 401:
                raise
            user = None
        return {'mode':mode, 'user':user}

    @app.post('/auth/login')
    def login(payload: Login, request: Request, response: Response):
        if mode != 'session':
            raise HTTPException(404)
        same_origin(request)
        # Persistent per-client rate limit; never trust forwarded client IP headers.
        bucket = hashlib.sha256((request.client.host if request.client else 'local').encode()).hexdigest()
        with psycopg.connect(dsn, row_factory=dict_row) as conn:
            conn.execute('DELETE FROM _specton_login_attempts WHERE resets_at<now()')
            limit = conn.execute('INSERT INTO _specton_login_attempts VALUES(%s,1,now()+interval \'1 minute\') '
                                 'ON CONFLICT(bucket) DO UPDATE SET attempts=_specton_login_attempts.attempts+1 RETURNING attempts', (bucket,)).fetchone()
        if limit['attempts'] > 10:
            raise HTTPException(429, detail='LOGIN_RATE_LIMIT', headers={'Retry-After':'60'})
        with psycopg.connect(dsn, row_factory=dict_row) as conn:
            row = conn.execute('SELECT * FROM _specton_users WHERE username=%s FOR UPDATE', (payload.username,)).fetchone()
            valid = verify_password(payload.password, row['password_hash'] if row else dummy)
            if not valid or not row or not row['active']:
                raise HTTPException(401, detail='INVALID_CREDENTIALS')
            conn.execute('DELETE FROM _specton_sessions WHERE expires_at<now()')
            token, csrf = secrets.token_urlsafe(32), secrets.token_urlsafe(32)
            conn.execute('INSERT INTO _specton_sessions VALUES(%s,%s,%s,now()+interval \'8 hours\')',
                         (hashlib.sha256(token.encode()).hexdigest(), row['username'], csrf))
            old = request.cookies.get(COOKIE)
            if old:
                conn.execute('DELETE FROM _specton_sessions WHERE token_hash=%s', (hashlib.sha256(old.encode()).hexdigest(),))
        response.set_cookie(COOKIE, token, httponly=True, secure=secure, samesite='strict', max_age=28800, path='/')
        response.headers['Cache-Control'] = 'no-store'
        return {'mode':mode, 'user':{'username':row['username'], 'roles':sorted(set(row['roles']) & approved_roles), 'csrf':csrf}}

    @app.post('/auth/logout', status_code=204)
    def logout(request: Request, response: Response):
        user = identity(request)
        if mode != 'session':
            raise HTTPException(404)
        same_origin(request)
        if not hmac.compare_digest(request.headers.get('x-csrf-token','').encode(), user['csrf'].encode()):
            raise HTTPException(403, detail='CSRF_REJECTED')
        with psycopg.connect(dsn) as conn:
            conn.execute('DELETE FROM _specton_sessions WHERE token_hash=%s',
                         (hashlib.sha256(request.cookies[COOKIE].encode()).hexdigest(),))
        response.delete_cookie(COOKIE, path='/', secure=secure, httponly=True, samesite='strict')
        response.headers['Cache-Control'] = 'no-store'

    return principal
