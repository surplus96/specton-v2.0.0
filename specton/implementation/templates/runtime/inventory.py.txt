"""Approved stock-ledger-v1. Decimal quantities and one transaction per command."""
import hashlib
import json
import re
from decimal import Decimal
from typing import Annotated, Literal

import psycopg
from fastapi import Header, HTTPException, Request
from psycopg import sql
from psycopg.rows import dict_row
from pydantic import AfterValidator, BaseModel, ConfigDict, Field, StrictStr

Key = Annotated[StrictStr, Field(min_length=1, max_length=64, pattern=r'^[A-Za-z0-9_-]+$')]
MAXIMUM = Decimal('999999999999999.999')


def quantity(value):
    if not re.fullmatch(r'(0|[1-9][0-9]{0,14})(\.[0-9]{1,3})?', value):
        raise ValueError('quantity must be a positive decimal string with at most 3 decimal places')
    number = Decimal(value)
    if not 0 < number <= MAXIMUM:
        raise ValueError('quantity out of range')
    return format(number, '.3f')


Quantity = Annotated[StrictStr, AfterValidator(quantity)]


class Payload(BaseModel):
    model_config = ConfigDict(extra='forbid')


class Draft(Payload):
    id: Key
    warehouse_id: Key
    item_id: Key
    kind: Literal['receive', 'issue']
    quantity: Quantity


class Edit(Payload):
    quantity: Quantity


class Empty(Payload):
    pass


class Movement(Payload):
    id: str
    warehouse_id: str
    item_id: str
    kind: Literal['receive', 'issue']
    quantity: str
    status: Literal['draft', 'confirmed', 'cancelled']


class Balance(Payload):
    warehouse_id: str
    item_id: str
    quantity: str


def install(conn, policy):
    warehouse, item = policy['warehouse'], policy['item']
    references = sql.SQL('FOREIGN KEY(warehouse_id) REFERENCES {}({}), FOREIGN KEY(item_id) REFERENCES {}({})').format(
        sql.Identifier(warehouse['table']), sql.Identifier(warehouse['column']),
        sql.Identifier(item['table']), sql.Identifier(item['column']))
    conn.execute(sql.SQL('''CREATE TABLE _specton_movements (
        id text PRIMARY KEY, warehouse_id text NOT NULL, item_id text NOT NULL,
        kind text NOT NULL CHECK(kind IN ('receive','issue')),
        quantity numeric(18,3) NOT NULL CHECK(quantity>0),
        status text NOT NULL CHECK(status IN ('draft','confirmed','cancelled')), {} )''').format(references))
    conn.execute(sql.SQL('''CREATE TABLE _specton_balances (
        warehouse_id text NOT NULL, item_id text NOT NULL,
        quantity numeric(18,3) NOT NULL CHECK(quantity>=0),
        PRIMARY KEY(warehouse_id,item_id), {} )''').format(references))
    conn.execute('''CREATE TABLE _specton_ledger (
        sequence bigserial PRIMARY KEY, movement_id text NOT NULL REFERENCES _specton_movements,
        event text NOT NULL CHECK(event IN ('confirm','cancel')),
        delta numeric(18,3) NOT NULL CHECK(delta<>0), actor_hash text NOT NULL,
        created_at timestamptz NOT NULL DEFAULT now(), UNIQUE(movement_id,event))''')


def require_references(conn, policy, warehouse_id, item_id):
    for name, value in [('warehouse', warehouse_id), ('item', item_id)]:
        ref = policy[name]
        found = conn.execute(sql.SQL('SELECT 1 FROM {} WHERE {}=%s').format(
            sql.Identifier(ref['table']), sql.Identifier(ref['column'])), (value,)).fetchone()
        if found is None:
            raise HTTPException(404, detail='REFERENCE_NOT_FOUND')


def movement(row):
    return {**row, 'quantity': format(row['quantity'], '.3f')}


def register(app, policy, dsn, principal):
    def execute(operation, request, payload, key, command_key):
        actor = principal(request, operation['actor_roles'])
        values = payload.model_dump()
        digest = hashlib.sha256(json.dumps({'key': key, 'payload': values}, sort_keys=True, separators=(',', ':')).encode()).hexdigest()
        scope = actor + ':' + operation['id']
        action = operation['action']
        with psycopg.connect(dsn, row_factory=dict_row) as conn:
            conn.execute('SELECT pg_advisory_xact_lock(hashtextextended(%s,0))', (scope + ':' + command_key,))
            previous = conn.execute('SELECT request_hash,result FROM _specton_commands WHERE scope=%s AND key=%s', (scope, command_key)).fetchone()
            if previous:
                if previous['request_hash'] != digest:
                    raise HTTPException(409, detail='IDEMPOTENCY_CONFLICT')
                return previous['result']
            if action == 'draft':
                require_references(conn, policy, payload.warehouse_id, payload.item_id)
                try:
                    row = conn.execute('''INSERT INTO _specton_movements VALUES(%s,%s,%s,%s,%s,'draft') RETURNING *''',
                        (payload.id,payload.warehouse_id,payload.item_id,payload.kind,Decimal(payload.quantity))).fetchone()
                except psycopg.errors.UniqueViolation:
                    raise HTTPException(409, detail='DUPLICATE_KEY') from None
            else:
                row = conn.execute('SELECT * FROM _specton_movements WHERE id=%s FOR UPDATE', (key,)).fetchone()
                if row is None:
                    raise HTTPException(404, detail='NOT_FOUND')
                if action == 'edit':
                    if row['status'] != 'draft':
                        raise HTTPException(409, detail='INVALID_STATE')
                    row = conn.execute('UPDATE _specton_movements SET quantity=%s WHERE id=%s RETURNING *', (Decimal(payload.quantity),key)).fetchone()
                else:
                    needed = 'draft' if action == 'confirm' else 'confirmed'
                    if row['status'] != needed:
                        raise HTTPException(409, detail='INVALID_STATE')
                    delta = row['quantity'] if row['kind'] == 'receive' else -row['quantity']
                    if action == 'cancel':
                        delta = -delta
                    pair = (row['warehouse_id'],row['item_id'])
                    conn.execute('INSERT INTO _specton_balances VALUES(%s,%s,0) ON CONFLICT DO NOTHING', pair)
                    balance = conn.execute('SELECT quantity FROM _specton_balances WHERE warehouse_id=%s AND item_id=%s FOR UPDATE', pair).fetchone()
                    amount = balance['quantity'] + delta
                    if amount < 0:
                        raise HTTPException(409, detail='INSUFFICIENT_STOCK')
                    if amount > MAXIMUM:
                        raise HTTPException(409, detail='QUANTITY_OVERFLOW')
                    conn.execute('UPDATE _specton_balances SET quantity=%s WHERE warehouse_id=%s AND item_id=%s', (amount,*pair))
                    conn.execute('INSERT INTO _specton_ledger(movement_id,event,delta,actor_hash) VALUES(%s,%s,%s,%s)', (key,action,delta,actor))
                    row = conn.execute('UPDATE _specton_movements SET status=%s WHERE id=%s RETURNING *', ('confirmed' if action=='confirm' else 'cancelled',key)).fetchone()
            result = movement(row)
            conn.execute('INSERT INTO _specton_commands VALUES(%s,%s,%s,%s)', (scope,command_key,digest,psycopg.types.json.Jsonb(result)))
            return result

    def bind(operation):
        action = operation['action']
        response = Movement
        if action == 'balance':
            def endpoint(request: Request, warehouse_id: Key, item_id: Key):
                principal(request, operation['actor_roles'])
                with psycopg.connect(dsn, row_factory=dict_row) as conn:
                    require_references(conn, policy, warehouse_id, item_id)
                    row = conn.execute('SELECT quantity FROM _specton_balances WHERE warehouse_id=%s AND item_id=%s', (warehouse_id,item_id)).fetchone()
                    return {'warehouse_id':warehouse_id,'item_id':item_id,'quantity':format(row['quantity'] if row else Decimal(0),'.3f')}
            response = Balance
        elif action == 'movement':
            def endpoint(key: Key, request: Request):
                principal(request, operation['actor_roles'])
                with psycopg.connect(dsn, row_factory=dict_row) as conn:
                    row = conn.execute('SELECT * FROM _specton_movements WHERE id=%s', (key,)).fetchone()
                    if row is None:
                        raise HTTPException(404, detail='NOT_FOUND')
                    return movement(row)
        elif action == 'draft':
            def endpoint(payload: Draft, request: Request, idempotency_key: Annotated[str, Header(min_length=1,max_length=128)]):
                return execute(operation,request,payload,None,idempotency_key)
        else:
            def endpoint(key: Key, payload, request: Request, idempotency_key: Annotated[str, Header(min_length=1,max_length=128)]):
                return execute(operation,request,payload,key,idempotency_key)
            endpoint.__annotations__['payload'] = Edit if action == 'edit' else Empty
        app.add_api_route(operation['path'],endpoint,methods=[operation['method']],status_code=operation['success_status'],
                          operation_id=operation['id'],response_model=response)

    for operation in policy['operations']:
        bind(operation)
